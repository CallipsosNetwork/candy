controller Auth {
  POST /signup -> Signup(email, password) {
    auth: none
    body: { email: Email, password: Password }
    map:
      ok(r)                    -> 201 { user_id: r.user }
      err(WeakPassword reason) -> 422 { error: "weak" }
  }
}

==================
controller with multiple routes
==================

controller Todos {
  GET /todos/:id -> Read(id) {
    auth: bearer
    map:
      ok(t) -> 200 t
  }

  DELETE /todos/:id -> Delete(id) {
    auth: bearer
    map:
      ok(_) -> 204
  }
}
