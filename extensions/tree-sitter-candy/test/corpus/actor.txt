actor Session(token: Token) {
  state {
    user: Id
    revoked: bool = false
  }

  invariant "expires > issued"

  accepts Revoke() -> unit {
    intent: "Revoke the session."
    effect: revoked = true
    emit SessionRevoked { token: self.token }
    commit
  }
}
