flow Signup(email: Email) -> Result<Id, EmailTaken> {
  intent: """Multi-line intent."""

  step user = ask User.create({ id: generate(), email })
              rescue compensate user; reject EmailTaken

  emit UserSignedUp { user: user.id }
  commit user.id
}

flow Login(email: Email) -> Result<Token, BadCreds> {
  step user = ask User.findBy(email)
  step ok   = if not verify(password) then reject BadCreds
  commit { user: user.id, token: session.token }
}
